### 核心问题

**问题 1：Agent 人设同质化（Phase 1）**
- 现象：不同 opinion_spreader 的 description 可能高度相似
- 影响：导致 Phase 3 发言时缺乏差异化

**问题 2：Agent 发言同质化（Phase 3）**
- 现象：同一群体的 Agent 发言内容几乎一模一样
- 影响：无法体现真实社交媒体的多样性

**问题 3：LLM 温度配置不合理**
- Phase 1 LLM2：温度未明确设置，可能过低导致生成内容保守
- Phase 3 发言：所有 Agent 使用统一温度，未区分事件实体和传播者

```python
def llm3_validate(llm2_output: Dict) -> Tuple[bool, str]:
    """
    LLM3 校验逻辑（v1.1.5 增强）
  
    校验内容：
    1. JSON 格式正确
    2. 双列实体（event_entities + opinion_spreaders）
    3. 总量 ≤ 15
    4. 立场分布合理（批评/中立/支持三个区间都有）
    5. 逻辑一致性（死忠粉 stance > 7，批评者 stance < 3）
    6. 人设差异化（description 不能高度相似）
  
    修改于：v1.1.5
    """
    # 现有校验逻辑...
  
    # ========== 新增校验 4：立场分布 ==========
    spreaders = llm2_output.get("opinion_spreaders", [])
    stances = [s["stance_score"] for s in spreaders]
  
    has_critic = any(s <= 3 for s in stances)      # 批评者
    has_neutral = any(4 <= s <= 6 for s in stances)  # 中立者
    has_supporter = any(s >= 7 for s in stances)   # 支持者
  
    if not (has_critic and has_neutral and has_supporter):
        return False, "立场分布不合理：必须同时包含批评者（1-3）、中立者（4-6）、支持者（7-10）"
  
    # ========== 新增校验 5：逻辑一致性 ==========
    for s in spreaders:
        group_name = s["group_name"]
        stance = s["stance_score"]
      
        # 检查支持类群体
        if any(keyword in group_name for keyword in ["死忠", "支持者", "粉丝"]):
            if stance < 7:
                return False, f"逻辑冲突：{group_name} 是支持类群体，但 stance={stance} < 7"
      
        # 检查批评类群体
        if any(keyword in group_name for keyword in ["批评", "维权", "反对"]):
            if stance > 3:
                return False, f"逻辑冲突：{group_name} 是批评类群体，但 stance={stance} > 3"
  
    # ========== 新增校验 6：人设差异化 ==========
    import difflib
    descriptions = [s["description"] for s in spreaders]
  
    # 计算两两相似度
    for i in range(len(descriptions)):
        for j in range(i+1, len(descriptions)):
            similarity = difflib.SequenceMatcher(
                None, descriptions[i], descriptions[j]
            ).ratio()
          
            if similarity > 0.7:  # 相似度超过 70% 视为重复
                return False, (
                    f"人设重复：以下两个群体的 description 相似度 {similarity:.2f} > 0.7\n"
                    f"- {spreaders[i]['group_name']}: {descriptions[i]}\n"
                    f"- {spreaders[j]['group_name']}: {descriptions[j]}"
                )
  
    return True, ""
```

```python
def generate_agent_comment(agent: GraphNode, saw_posts: List[Dict], ...) -> Dict:
    """
    生成 Agent 发言
  
    修改于：v1.1.5
    修改内容：差异化 LLM 温度（事件实体 vs 传播者）
    """
  
    prompt = build_agent_prompt(agent, saw_posts, ...)
  
    # 根据实体类型选择温度
    if agent.entity_category == "event_entity":
        temperature = config.PHASE3_EVENT_ENTITY_TEMPERATURE  # 0.3
    else:
        temperature = config.PHASE3_SPREADER_TEMPERATURE      # 0.8
  
    response = llm_client.call(
        prompt=prompt,
        model=config.LLM_MODEL_NAME,
        temperature=temperature  # ← 差异化温度
    )
  
    return response
```

```python
def check_comment_duplication(tick_entries: List[TickEntry]) -> List[str]:
    """
    检测同一 Tick 内是否有完全相同的发言
  
    新增于：v1.1.5
  
    返回：重复发言的列表（如果有）
    """
    comments = [entry.comment for entry in tick_entries]
    duplicates = []
  
    for i in range(len(comments)):
        for j in range(i+1, len(comments)):
            if comments[i] == comments[j]:
                duplicates.append(f"Agent {tick_entries[i].agent_id} 和 Agent {tick_entries[j].agent_id} 发言完全相同：{comments[i]}")
  
    return duplicates
```

**在 Tick 主循环中调用**：
```python
def run_tick(tick_num: int, agents: List[GraphNode], ...) -> TickOutput:
    """运行单个 Tick"""
  
    tick_entries = []
    for agent in agents:

        tick_entries.append(entry)
  
    # ========== 新增：发言去重检测 ==========

        logger.warning(f"[Tick {tick_num}] 检测到重复发言：")
        for dup in duplicates:
```
